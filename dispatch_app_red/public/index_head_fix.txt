    /* 深度優化：自動切換與持久化記憶邏輯 */
    (function() {
      const params = new URLSearchParams(window.location.search);
      const modeParam = params.get('mode');
      const savedMode = localStorage.getItem('preferred_view_mode');
      
      if (modeParam === 'desktop') {
        localStorage.setItem('preferred_view_mode', 'desktop');
        return;
      }
      
      const shouldBeMobile = modeParam === 'mobile' || 
                             savedMode === 'mobile' || 
                             (window.innerWidth < 1024 && savedMode !== 'desktop');

      if (shouldBeMobile) {
        if (modeParam === 'mobile') localStorage.setItem('preferred_view_mode', 'mobile');
        window.location.href = 'mobile.html' + window.location.search;
      }
    })();
  </script>
